% =========================================================
%                Academic CV PDF Header
% =========================================================

% ---------------------------------------------------------
% Engine & Font
% ---------------------------------------------------------
\usepackage{fontspec}
\setmainfont{Times New Roman}

% ---------------------------------------------------------
% Micro-typography & Spacing
% ---------------------------------------------------------
\usepackage{microtype}
\usepackage{ragged2e}
\usepackage{setspace}

\setlength{\parindent}{0pt}
\setlength{\parskip}{1pt}
\linespread{0.98}

% ---------------------------------------------------------
% Colors
% ---------------------------------------------------------
\usepackage{xcolor}

\definecolor{cvblue}{HTML}{1F4E79}
\definecolor{cvgray}{gray}{0.45}

% ---------------------------------------------------------
% Icons
% ---------------------------------------------------------
\usepackage{fontawesome5}
\newcommand{\icon}[1]{\textcolor{cvblue}{#1}}
\newcommand{\cvheader}[3]{%
  \begin{minipage}[t]{0.62\textwidth}
    {\LARGE\bfseries #1}\par
    \vspace{0.15em}
    {\small #2}
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{0.34\textwidth}
    \raggedleft\small #3
  \end{minipage}
  \vspace{0.45em}
  \hrule
  \vspace{0.7em}
}

% ---------------------------------------------------------
% Layout
% ---------------------------------------------------------
\usepackage{geometry}
\geometry{margin=0.62in}

\setlength{\columnsep}{16pt}
\setlength{\columnseprule}{0pt}

% ---------------------------------------------------------
% Lists (tight academic CV style)
% ---------------------------------------------------------
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.3em, itemsep=0pt, parsep=0pt, topsep=2pt}
\setlist[enumerate]{leftmargin=1.5em, itemsep=0pt, parsep=0pt, topsep=2pt}

% ---------------------------------------------------------
% Section Formatting
% ---------------------------------------------------------
\usepackage{titlesec}

\titleformat{\section}
  {\normalfont\large\bfseries}
  {}
  {0pt}
  {}
  [\vspace{-0.25em}\titlerule]

\titlespacing*{\section}
  {0pt}
  {0.75\baselineskip}
  {0.35\baselineskip}

\titleformat{\subsection}
  {\normalfont\normalsize\bfseries}
  {}
  {0pt}
  {}

\titlespacing*{\subsection}
  {0pt}
  {0.45\baselineskip}
  {0.1\baselineskip}

\titleformat{\subsubsection}
  {\normalfont\normalsize\itshape}
  {}
  {0pt}
  {}

\titlespacing*{\subsubsection}
  {0pt}
  {0.35\baselineskip}
  {0.1\baselineskip}

% ---------------------------------------------------------
% Title Formatting
% ---------------------------------------------------------
\usepackage{titling}

\pretitle{\begin{center}\Huge\bfseries}
\posttitle{\par\end{center}\vspace{-1.2em}}

\predate{\begin{center}}
\postdate{\par\end{center}\vspace{-1em}}

% ---------------------------------------------------------
% Links
% ---------------------------------------------------------
\usepackage[hidelinks]{hyperref}
\hypersetup{
  colorlinks=true,
  urlcolor=cvblue
}

% ---------------------------------------------------------
% Footer Style (for last updated line if needed)
% ---------------------------------------------------------
\newcommand{\cvupdated}[1]{
  \vspace{1em}
  \begin{center}
    {\color{cvgray}\textit{#1}}
  \end{center}
}

% ---------------------------------------------------------
% Clean Page
% ---------------------------------------------------------
\pagestyle{empty}
