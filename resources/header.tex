% Teaching table: flexible course column and light horizontal rules.
\usepackage{tabularx}
\usepackage{booktabs}

% Template-style experience and education entries. Each heading stays on one page;
% long institution names wrap without colliding with the right-aligned dates.
\newcommand{\cvexperience}[4]{%
  \par\addvspace{2pt}\noindent
  \begingroup\small
  \setlength{\tabcolsep}{4pt}%
  \renewcommand{\arraystretch}{1}%
  \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}Xr@{}}
    \textbf{#1} & #2 \\
    \textit{#3} & \textit{#4}
  \end{tabularx}\par
  \endgroup
}

\usepackage{needspace}

% Description bullets under an experience entry: indented past the entry's
% text, with a little air between bullets and before the next entry.
\newenvironment{cvdetails}{%
  \begin{itemize}[leftmargin=2.6em, labelsep=0.5em, topsep=1pt, itemsep=1pt, parsep=0pt, partopsep=0pt]
  \small
}{%
  \end{itemize}\vspace{2pt}
}

% French typography would replace the list bullets with long dashes; keep the
% same round bullets as the English CV.
\ifdefined\frenchsetup
  \frenchsetup{StandardLists=true}
\fi

% Closing line: the link to the CV in the other language, small and gray, in the
% footer of the last page (so it never pushes the CV onto an extra page).
\usepackage{fancyhdr}
% The page style is defined once, globally; \cvotherlanguage only stores the
% text (globally too), so it also works inside a section's indentation group.
\newcommand{\cvotherlanguagetext}{}
\newcommand{\cvotherlanguageurl}{}
\fancypagestyle{cvlastpage}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \fancyfoot[C]{\footnotesize\color{cvgray}%
    \ifdefined\NoAutoSpacing\NoAutoSpacing\fi
    \cvotherlanguagetext~\href{\cvotherlanguageurl}{\textcolor{cvgray}{\nolinkurl{\cvotherlanguageurl}}}}%
}
\newcommand{\cvotherlanguage}[2]{%
  \gdef\cvotherlanguagetext{#1}%
  \gdef\cvotherlanguageurl{#2}%
  \thispagestyle{cvlastpage}%
}

% One-page CV: dated rows (a narrow date column, then the text) and compact
% section titles with a full-width rule. Only the one-page CV uses these.
\newlength{\cvdatewidth}
\setlength{\cvdatewidth}{1.95cm}
\newenvironment{cvdated}{%
  \par\small
  \setlength{\tabcolsep}{0pt}%
  \renewcommand{\arraystretch}{1.02}%
  \noindent
  \tabularx{\linewidth}{@{}>{\raggedright\arraybackslash}p{\cvdatewidth}@{\hspace{0.6em}}>{\raggedright\arraybackslash}X@{}}%
}{%
  \endtabularx\par
}
\newcommand{\cvcompactlayout}{%
  \titleformat{\subsection}[block]{\normalfont\normalsize\bfseries}{}{0pt}{}[\vspace{0.5pt}\titlerule]%
  \titlespacing*{\subsection}{0pt}{0.3\baselineskip}{0.15\baselineskip}%
}

% ---------------------------------------------------------
% Lists (tight academic CV style)
% ---------------------------------------------------------
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.3em, itemsep=0pt, parsep=0pt, topsep=2pt}
\setlist[enumerate]{leftmargin=1.5em, itemsep=0pt, parsep=0pt, topsep=2pt}

% Each heading's contents move one step to the right. The PDF-only Pandoc
% filter nests these environments, so wrapped lines keep the same margin.
\newlength{\cvindentstep}
\setlength{\cvindentstep}{1.8em}
\newenvironment{cvindent}{%
  \begin{list}{}{%
    \setlength{\leftmargin}{\cvindentstep}%
    \setlength{\rightmargin}{0pt}%
    \setlength{\topsep}{0pt}%
    \setlength{\partopsep}{0pt}%
    \setlength{\parsep}{\parskip}%
    \setlength{\itemsep}{0pt}%
  }%
  \item\relax
  % The enclosing environment already supplies the content's indentation.
  \setlist[itemize,1]{leftmargin=0pt}%
  \setlist[enumerate,1]{leftmargin=0pt}%
}{%
  \end{list}%
}
